1 Все это можно делать через указатели 

int arr[5] = {10, 20, 30, 40, 50};

int* it  = arr;       // stands on the first element
int* end = arr + 5;   // stands PAST the last element (not on 50!)

while (it != end) {
    std::cout << *it << " ";   // read the element
    ++it;                      // move to the next
}
// 10 20 30 40 50

операции которые тут есть 
итератор — это любой объект, который умеет эти три операции.
*it	дай элемент, на котором стоишь
++it	перейди к следующему
it != end	мы ещё не дошли до конца?

2 Во что разворачивается range for loop 
for (int x : container) {
    std::cout << x;
}
{
    auto it  = container.begin();
    auto end = container.end();
    for (; it != end; ++it) {
        int x = *it;
        std::cout << x;
    }
}

Отсюда полный список требований range-for:

у container есть begin() и end();
они возвращают объекты одного типа (итераторы);
у итератора есть !=, ++ и *.

3 Совсем простой пример
struct MyArray {
    int data[5] = {10, 20, 30, 40, 50};

    int* begin() { return data; }       // first element
    int* end()   { return data + 5; }   // past the last
};

MyArray a;
for (int x : a)
    std::cout << x << " ";   // 10 20 30 40 50
Мы не писали никакого итератора: указатель уже умеет *, ++ и !=.

4 Более сложный 
template<typename T>
class StepIterator {
    T* p;      // where we are
    int step;  // how far to jump

public:
    StepIterator(T* p, int step) : p(p), step(step) {}

    T& operator*() const {                 // reference: the element exists in memory, we can write to it
        return *p;
    }
    StepIterator& operator++() {
        p += step;                         // the only difference from a plain pointer
        return *this;
    }
    bool operator!=(const StepIterator& other) const {
        return p != other.p;
    }
};

template<typename T>
struct StridedView {
    T* data;
    int length;   // number of elements we visit
    int step;

    StepIterator<T> begin() const { return StepIterator<T>(data, step); }
    StepIterator<T> end() const   { return StepIterator<T>(data + length * step, step); }  // exactly reachable
};

int arr[8] = {0, 1, 2, 3, 4, 5, 6, 7};
StridedView<int> even{arr, 4, 2};

for (int& x : even)   // int& - we write through the iterator
    x = -1;
// arr: -1 1 -1 3 -1 5 -1 7
