До с++11 было несколько типов инициализации
    int x = 5;             // copy initialization
    int y(5);              // direct init
    int arr[] = {1, 2, 3}; // aggregate init
    Widget w(1, 2);        // Constructor call

int a[3](1, 2, 3);   // ОШИБКА: массив нельзя инициализировать круглыми скобками
int a[3]{1, 2, 3};   // ОК

struct Point { int x, y; };
Point p(1, 2);       // ОШИБКА (до C++20): у Point нет конструктора
Point p{1, 2};       // ОК всегда

class Widget {
    int data[3];
public:
    Widget() : data(1, 2, 3) {}   // ОШИБКА: так нельзя
    Widget() : data = {1, 2, 3} {} // ОШИБКА: и так нельзя
    Widget() : data{1, 2, 3} {}    // ОК — только фигурные скобки
};
